import os
import json
from whisper import load_model

input_directory = r"C:\Users\User\OneDrive - University of Helsinki\Desktop\Vaibhav_Studies\6th Teaching Period\HCI Project\Videos"

output_json_file = "transcription_results.json"

model = load_model("base")

transcription_results = []

video_extensions = {".mp4", ".avi", ".mov", ".mkv", ".flv"}

for root, _, files in os.walk(input_directory):
    for file in files:
        file_path = os.path.join(root, file)
        file_extension = os.path.splitext(file)[-1].lower()

        # Skip non-video files and JSON files
        if file_extension in video_extensions:
            print(f"Processing file: {file_path}")
            try:
                # Transcribe the video file
                result = model.transcribe(file_path)

                # Append the result to the transcription results
                transcription_results.append({
                    "file_name": file,
                    "full_path": file_path,
                    "transcription": result.get("text", "")
                })

            except Exception as e:
                print(f"Failed to process {file_path}: {e}")

with open(output_json_file, "w", encoding="utf-8") as f:
    json.dump(transcription_results, f, indent=4, ensure_ascii=False)

print(f"Transcription completed. Results saved in {output_json_file}")
