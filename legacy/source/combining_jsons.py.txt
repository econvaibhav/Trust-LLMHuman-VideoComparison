import os
import json

transcription_english_file = "transcription_results_english.json"
summary_file = "video_summaries.json"
metadata_file = r"C:\Users\User\OneDrive - University of Helsinki\Desktop\Vaibhav_Studies\Future_Work_Datacollection\Data\political_shorts_all_data.json"
output_combined_file = "combined_english_data.json"

with open(transcription_english_file, "r", encoding="utf-8") as f:
    english_transcriptions = json.load(f)

with open(summary_file, "r", encoding="utf-8") as f:
    video_summaries = json.load(f)

with open(metadata_file, "r", encoding="utf-8") as f:
    metadata = json.load(f)

# Map metadata using titles to make it compatible with file names
metadata_map = {entry["title"].strip(): entry for entry in metadata}
summary_map = {video["file_name"]: video for video in video_summaries}

combined_data = []
for video in english_transcriptions:
    file_name = video["file_name"]
    title = os.path.splitext(file_name)[0].strip()
    metadata_entry = metadata_map.get(title, {})
    summary = summary_map.get(file_name, {}).get("summary", "No summary available")
    
    combined_entry = {
        "file_name": file_name,
        "full_path": video["full_path"],
        "transcription": video["transcription"],
        "summary": summary,
        "metadata": metadata_entry.get("metadata", {}),
        "comments": metadata_entry.get("comments", [])
    }
    combined_data.append(combined_entry)

with open(output_combined_file, "w", encoding="utf-8") as f:
    json.dump(combined_data, f, indent=4, ensure_ascii=False)

print(f"Combined data saved in {output_combined_file}")
